plot_one_gene <- function(g){
  df <- exp_long %>% dplyr::filter(Gene == g) %>% droplevels()
  
  # 保证分组顺序
  df$group <- factor(df$group, levels = c("High","Low"))
  
  y_min <- min(df$Expression, na.rm = TRUE)
  y_max <- max(df$Expression, na.rm = TRUE)
  y_pad <- max(0.06 * (y_max - y_min), 0.25)   # 横线与图顶的间距
  
  # 只有当两组都存在时才做检验+标注
  has_two_groups <- all(c("High","Low") %in% levels(droplevels(df$group))) &&
    length(unique(na.omit(df$group))) == 2
  
  if (has_two_groups) {
    pval <- tryCatch(
      wilcox.test(Expression ~ group, data = df)$p.value,
      error = function(e) NA_real_
    )
    p_lab <- if (is.na(pval)) "NA" else if (pval < 1e-4) "<0.0001" else paste0("P = ", signif(pval, 3))
    
    # ✅ 用 group1 / group2（不要 xmin/xmax）
    df_bracket <- data.frame(
      group1 = "High",
      group2 = "Low",
      y.position = y_max + y_pad,
      label = p_lab
    )
  }
  
  p <- ggplot(df, aes(group, Expression, fill = group)) +
    geom_violin(trim = FALSE, width = 0.9, alpha = 0.95, color = "black") +
    geom_boxplot(width = 0.16, fill = "white", color = "black", outlier.shape = NA) +
    scale_fill_manual(values = pal, guide = "none") +
    labs(x = NULL, y = g) +                 # Y 轴标题=基因名
    theme_classic(base_size = 14) +
    scale_y_continuous(expand = expansion(mult = c(0.02, 0.20))) +
    theme(
      plot.title  = element_blank(),
      axis.title.x = element_blank(),
      axis.text.x  = element_text(size = 12),
      axis.line.x  = element_line(color = "black"),
      axis.ticks.x = element_line(color = "black"),
      axis.line.y  = element_line(color = "black"),
      axis.ticks.y = element_line(color = "black"),
      axis.text.y  = element_text(size = 12),
      axis.title.y = element_text(size = 13, face = "bold")
    )
  
  # 只有两组都在时再叠加横线+P值
  if (has_two_groups) {
    p <- p + stat_pvalue_manual(
      df_bracket,
      label = "label",
      y.position = "y.position",
      tip.length = 0, bracket.size = 0.6, size = 4,
      inherit.aes = FALSE
    )
  }
  
  return(p)
}

plot_one_gene <- function(g){
  df <- exp_long %>% dplyr::filter(Gene == g) %>% droplevels()
  
  # 保证分组顺序
  df$group <- factor(df$group, levels = c("High","Low"))
  
  y_min <- min(df$Expression, na.rm = TRUE)
  y_max <- max(df$Expression, na.rm = TRUE)
  y_pad <- max(0.06 * (y_max - y_min), 0.25)   # 横线与图顶的间距
  
  # 只有当两组都存在时才做检验+标注
  has_two_groups <- all(c("High","Low") %in% levels(droplevels(df$group))) &&
    length(unique(na.omit(df$group))) == 2
  
  if (has_two_groups) {
    pval <- tryCatch(
      wilcox.test(Expression ~ group, data = df)$p.value,
      error = function(e) NA_real_
    )
    p_lab <- if (is.na(pval)) "NA" else if (pval < 1e-4) "<0.0001" else paste0("P = ", signif(pval, 3))
    
    # ✅ 用 group1 / group2（不要 xmin/xmax）
    df_bracket <- data.frame(
      group1 = "High",
      group2 = "Low",
      y.position = y_max + y_pad,
      label = p_lab
    )
  }
  
  p <- ggplot(df, aes(group, Expression, fill = group)) +
    geom_violin(trim = FALSE, width = 0.9, alpha = 0.95, color = "black") +
    geom_boxplot(width = 0.16, fill = "white", color = "black", outlier.shape = NA) +
    scale_fill_manual(values = pal, guide = "none") +
    labs(x = NULL, y = g) +                 # Y 轴标题=基因名
    theme_classic(base_size = 14) +
    scale_y_continuous(expand = expansion(mult = c(0.02, 0.20))) +
    theme(
      plot.title  = element_blank(),
      axis.title.x = element_blank(),
      axis.text.x  = element_text(size = 12),
      axis.line.x  = element_line(color = "black"),
      axis.ticks.x = element_line(color = "black"),
      axis.line.y  = element_line(color = "black"),
      axis.ticks.y = element_line(color = "black"),
      axis.text.y  = element_text(size = 12),
      axis.title.y = element_text(size = 13, face = "bold")
    )
  
  # 只有两组都在时再叠加横线+P值
  if (has_two_groups) {
    p <- p + stat_pvalue_manual(
      df_bracket,
      label = "label",
      y.position = "y.position",
      tip.length = 0, bracket.size = 0.6, size = 4,
      inherit.aes = FALSE
    )
  }
  
  return(p)
}

stat_compare_means()：添加 p 值，默认使用 wilcox.test，你也可以改为 t.test。

palette = "jco"：优雅配色方案之一，你还可以试试 "npg", "lancet", "nejm" 等。
